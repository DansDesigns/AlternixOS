# How to create ISO on Windows with WSL2

# to change the location of the folder, change the line:
# cp -r /mnt/[DRIVE]/[FOLDER/TO/PATH]/AlternixOS/alternix-installer-gui


# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# 1.
# Type (or copy down to the "sudo bash build-iso.sh" command) the following lines into Debian WSL2:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

cd ~

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# remove folder if it already exists (if re-building):
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

sudo rm -rf ~/alternix-installer

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# copy installer folder from Windows file system to Debian WSL2 (replace "Users/dan/github" with your username & folder location):
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

cp -r /mnt/c/Users/dan/github/AlternixOS/alternix-installer-gui ~/alternix-installer

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# sometimes .efi file isn't copied, if so uncomment:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

#cp -r /mnt/c/Users/dan/GitHub/AlternixOS/alternix-installer-gui/efi ~/alternix-installer/efi


cd ~/alternix-installer

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# import working visor_x64.efi:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

./import-working-visor.sh ~/alternix-installer/visor_x64.efi

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# x86_64 Build Command:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

sudo env DEVUAN_MIRROR=https://ftp.fau.de/devuan/merged bash build-iso.sh

#sudo bash build-iso.sh



# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# ARM64 Build Command:
# install these first:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~


sudo apt install qemu-user-static binfmt-support debootstrap dosfstools fdisk xz-utils

sudo bash build-arm.sh



# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# 2.
# ISO appears in ~/alternix-installer
# Copy it back to windows file system when done (replace "dan/github" with your username & folder location):
# 
#==========================================================
# /!\ New update to the iso builder now has 2 options /!\
#==========================================================
#
# x86-64 net-install
# x86-64 live preview
# The generated files will be named alternix-net-install.iso and alternix-live-preview.iso,
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

# net install:

cp ~/alternix-installer/alternix-net-install.iso /mnt/c/Users/dan/github/AlternixOS/alternix-net-install.iso

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

# live preview: 

cp ~/alternix-installer/alternix-live-preview.iso /mnt/c/Users/dan/github/AlternixOS/alternix-live-preview.iso

# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# import working visor_x64.efi:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

./import-working-visor.sh ~/alternix-installer/visor_x64.efi



# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
#logs:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

cp build/lb-build.log /mnt/c/Users/dan/GitHub/AlternixOS/alternix-installer-gui/lb-build.log
cp build/lb-build-attempts.log /mnt/c/Users/dan/GitHub/AlternixOS/alternix-installer-gui/lb-build-attempts.log


# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# 32bit uefi file:
# ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~

cp -r /mnt/c/Users/dan/GitHub/AlternixOS/alternix-installer-gui/efi ~/alternix-installer/efi
