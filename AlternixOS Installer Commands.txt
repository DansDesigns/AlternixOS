# How to create ISO on Windows with WSL2

# unzip nexos-installer.zip
# to change the location of NexOS folder, change the line:
# cp -r /mnt/[DRIVE]/[FOLDER-PATH]/NexOS/nexos-installer

# 1. Type the following lines into Debian WSL2:
~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~


cd ~

# remove folder if it already exists (if re-building):
sudo rm -rf ~/alternix-installer

# copy installer folder from Windows file system to Debian WSL2 (replace "Users/dan/github" with your username & folder location):
cp -r /mnt/c/Users/dan/github/AlternixOS/alternix-installer-gui ~/alternix-installer


cd ~/alternix-installer

sudo bash build-iso.sh


~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
# (future upgrade) specific build flags:
sudo bash build-iso.sh --arm    # ARM64
sudo bash build-iso.sh --amd    # AMD64 



# 2. ISO appears in ~/alternix-installer/


# 3. Copy it back to windows file system when done (replace "dan/github" with your username & folder location):

cp ~/alternix-installer/alternix-installer.iso /mnt/c/Users/dan/github/AlternixOS/alterniix-installer-gui.iso

(future versions)
cp ~/alternix-installer/alternix-installer-amd64.iso /mnt/c/Users/dan/github/AlternixOS/
cp ~/alternix-installer/alternix-installer-arm64.iso /mnt/c/Users/dan/github/AlternixOS/


ERROR NOTE:
When you drop to shell after Ctrl+C, the commands to know are:

cat /tmp/nexos-install.log | tail -50 — main installer log
cat /tmp/sel4-build.log | tail -50 — seL4 build detail
bash /installer/install.sh — restart installer manually
